# (Original command 0x4D)
CMD 0x4D
DATA 0x55

# PASSWORD
CMD 0xF3
DATA 0x0A

# panel setting
CMD 0x00
DATA 0xFF 0x09

# Power setting Register
CMD 0x01
DATA 0x03 0x00 0x3F 0x3F 0xA6

# BOOST ,驱动强度设定
CMD 0x06
DATA 0xC7 0xC7 0x2F 0xA5 0x33


# Frame rate (Register 0x30)
CMD 0x30
DATA 0x13

# VCOM and DATA interval setting Register
CMD 0x50
DATA 0x97

# 128S*250G
CMD 0x61
DATA 0xC8 0xC8

# VCOM
CMD 0x82
DATA 0x0A # 原始代码使用变量 VCOM_BUF (注释: VCOM=-1.3V 0C)

# Power Saving Register
CMD 0xE3
DATA 0x00

# LUT Option Register
CMD 0x2A
DATA 0x00 0x00 0x00

# 上电 (Power on)
CMD 0x04
WAIT_BUSY # 对应 C 代码中的 lcd_chkstatus()
