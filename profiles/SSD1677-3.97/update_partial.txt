# 局部刷新脚本
# 1. 设置RAM地址指针
CMD 0x4E
DATA 0x00
CMD 0x4F
DATA 0x00 0x00

# 2. 写入新的图像数据 (前置命令: 0x24)
CMD 0x24
#    <上位机将在此处发送'new'图像缓冲区的数据>
#    注意: C代码中对pic[]数据进行了按位取反(~), 这个逻辑需要上位机在处理图像时完成

# 3. 写入旧的图像数据 (前置命令: 0x26)
CMD 0x26
#    <上位机将在此处发送'old'图像缓冲区的数据>

# 4. 触发刷新
CMD 0x22
DATA 0xCF

CMD 0x20
WAIT_BUSY_SSD
