# UC8179 初始化脚本

# Panel setting
CMD 0x00
DATA 0x3F 

# Power setting
CMD 0x01
DATA 0x07 0x17 0x3F 0x3F 0x00

# Booster soft start
CMD 0x06
DATA 0x2d 0x2f 0x28 0x17

# LUT setting
CMD 0x2A
DATA 0x00 0x00

# Resolution setting: 128x250
CMD 0x61
DATA 0x03 0x20 0x01 0xE0 

# TCON setting
CMD 0x65
DATA 0x00 0x00 0x00 0x00

# TCON setting
CMD 0x60
DATA 0x22

# OSC setting: 100Hz
CMD 0x30
DATA 0x06

# VCOM setting
CMD 0x82 
DATA 0x09 # 使用一个通用的VCOM默认值


# Border waveform control
CMD 0x50
DATA 0x29 0x07

# Power on
CMD 0x04
WAIT_BUSY
