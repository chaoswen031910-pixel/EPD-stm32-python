# UC8251D 初始化脚本

# Panel setting
CMD 0x00
DATA 0xBF 0x0D

# Power setting
CMD 0x01
DATA 0x03 0x00 0x3F 0x3F 0x00

# Booster soft start
CMD 0x06
DATA 0x17 0x17 0x16

# LUT setting
CMD 0x2A
DATA 0x00 0x00 0x00 0xFF 0x03

# Resolution setting: 128*250
CMD 0x61
DATA 0x80      # 宽度: 128
DATA 0x00 0xFA # 高度: 250

# TCON setting
CMD 0x60
DATA 0x22

# OSC setting: 100Hz
CMD 0x30
DATA 0x13

# VCOM setting
CMD 0x82
DATA 0x08 # 使用一个通用的VCOM默认值, 0x1A

# Power off sequence setting
CMD 0x03
DATA 0x00

# Border waveform control
CMD 0x50
DATA 0xB7 0x04

# Power on
CMD 0x04
WAIT_BUSY
