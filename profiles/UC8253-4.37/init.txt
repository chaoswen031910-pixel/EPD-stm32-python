# UC8253c 初始化脚本

# Panel setting
CMD 0x00
DATA 0xFF 0x0D

# Power setting
CMD 0x01
DATA 0x03 0x10 0x3F 0x3F 0x00

# Booster soft start
CMD 0x06
DATA 0x17 0x17 0x16

# Resolution setting: 128x250
CMD 0x61
DATA 0xB0      # 宽度: 176
DATA 0x01 0xE0 # 高度: 480

# TCON setting
CMD 0x60
DATA 0x11

# OSC setting: 100Hz
CMD 0x30
DATA 0x13

# VCOM setting
CMD 0x82
DATA 0x08 # 使用一个通用的VCOM默认值, 0x1A

# Power off sequence setting
CMD 0x03
DATA 0x00

# Border waveform control
CMD 0x50
DATA 0xB7

# Power on
CMD 0x04
WAIT_BUSY
