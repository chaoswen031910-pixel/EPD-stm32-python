# panel setting
CMD 0x00
DATA 0x3F 0x0D

# power setting
CMD 0x01
DATA 0x03 0x10 0x3F 0x3F 0x0D

# booster start
CMD 0x06
DATA 0x28 0x28 0x16

# 配置波形的相关参数
CMD 0x2A
DATA 0x80 0x00 0x00 0xFF 0x00

# GATE/SOURCE START SETTING
CMD 0x65
DATA 0x00 0x00 0x00 0x00

# resolution setting (400x300)
CMD 0x61
DATA 0x01 0x90 0x01 0x2C

# PLL control
CMD 0x30
DATA 0x13

# TCON
CMD 0x60
DATA 0x00 0x13

# VCOM_DC
CMD 0x82
DATA 0x0A

# setting power off sequence
CMD 0x03
DATA 0x00

# (Register 0x50)
CMD 0x50
DATA 0x97 0x00
